// icon-color: red; icon-glyph: dumbbell;

// ============================================================================
// Today's Workout — a home-screen widget for the iOS Scriptable app
// ============================================================================
//
// Shows today's steps, push-ups and leg raises against their goals, with a
// coloured progress bar for each, the current streak, and a motivational line.
// Small, Medium and Large are all supported — the layout changes shape rather
// than just shrinking, because a Large layout scaled into a Small widget is
// unreadable.
//
// SETUP
//   1. Scriptable → + → paste this whole file.
//   2. Set API_KEY in the CONFIG block below to the same value as the
//      WORKOUT_API_KEY environment variable on the server. Nothing else needs
//      to change.
//   3. Long-press the home screen → add a Scriptable widget → any size →
//      choose this script.
//
// WIDGET PARAMETER (optional)
//   The widget's "Parameter" field takes a comma-separated list of tokens, in
//   any order, and each one overrides a CONFIG value for that widget only:
//
//     p1 | p2                which athlete to show
//     small | medium | large force a layout (otherwise the real size is used)
//     dark | clear           force an appearance — see APPEARANCE below
//
//   e.g. `p1, clear` — Abby's numbers, glass styling, size taken from iOS.
//
// ON CLEAR / TINTED ICONS
//   When the home screen uses the Clear or Tinted icon style, iOS applies its
//   own treatment to widgets, and a script cannot read or opt out of it — there
//   is no API that reports the icon style, so `APPEARANCE` is a manual switch
//   rather than something detected.
//
//   What this file controls is the artwork underneath: `APPEARANCE = "clear"`
//   drops the solid black slab, which is what looks like a hole punched through
//   a glass home screen, for a thin translucent scrim with shadowed text. It
//   degrades safely — if the platform composites it over an opaque background
//   instead of the wallpaper, it just reads as a lighter card.
//
//   The layouts also avoid depending on colour alone: each exercise carries an
//   emoji and a label as well as a hue, so three bars stay distinguishable if
//   the system desaturates them.
//
// ON ANIMATION
//   iOS home-screen widgets are static snapshots rendered by WidgetKit. Neither
//   Scriptable nor any other widget can animate them — there is no render loop
//   to animate in, and `presentLarge()` is a still image too. So the "smooth"
//   part here is visual rather than moving: a gradient background, rounded
//   geometry, and bars whose fills are drawn at sub-pixel precision. The widget
//   refreshes itself roughly every 15 minutes, which is the closest thing to
//   motion the platform offers.
//
// ============================================================================

/**

 *

/** A non-negative integer, whatever the API sent. Guards against null/NaN. */
function safeNum(value, fallback = 0) {
  const n = Number(value);
  return Number.isFinite(n) && n >= 0 ? Math.round(n) : fallback;
}

/** Percent complete, clamped to 0–100. A zero goal counts as done. */
function pct(value, goal) {
  if (!goal || goal <= 0) return 100;
  return Math.max(0, Math.min(100, (value / goal) * 100));
}

/** 8423 → "8,423" */
function fmt(n) {
  return safeNum(n).toLocaleString("en-US");
}

/** Message keyed off overall progress. */
function motivation(overall, allComplete) {
  if (allComplete || overall >= 100) return "Goals completed! 🎉";
  if (overall >= 67) return "Almost there!";
  if (overall >= 34) return "Keep going!";
  return "Great start!";
}

/** Local wall-clock time as HH:mm. */
function nowHHmm() {
  const df = new DateFormatter();
  df.dateFormat = "HH:mm";
  return df.string(new Date());
}

/** The IANA zone of this device, so the server can resolve "today" correctly. */
function deviceTimeZone() {
  try {
    return Intl.DateTimeFormat().resolvedOptions().timeZone || "UTC";
  } catch (e) {
    return "UTC";
  }
}

/** Widget "Parameter" field split into tokens, e.g. `p1, small` → ["p1","small"]. */
function parameterTokens() {
  return String(args.widgetParameter || "")
    .toLowerCase()
    .split(/[,\s]+/)
    .filter(Boolean);
}

/** "p1"/"p2" from the widget's Parameter field, else the PERSON constant. */
function resolvePerson() {
  const tokens = parameterTokens();
  if (tokens.indexOf("p1") !== -1) return "p1";
  if (tokens.indexOf("p2") !== -1) return "p2";
  return PERSON;
}

/**
 * Which of the three layouts to build.
 *
 * `config.widgetFamily` is set only while iOS is rendering a real home-screen
 * widget; running the script by hand in the app leaves it undefined. A size
 * token in the Parameter field therefore wins — that's also how you preview the
 * small and medium layouts without adding them to the home screen first.
 */
function resolveFamily() {
  const tokens = parameterTokens();
  if (tokens.indexOf("small") !== -1) return "small";
  if (tokens.indexOf("medium") !== -1) return "medium";
  if (tokens.indexOf("large") !== -1) return "large";

  const family = config.widgetFamily;
  if (family === "small" || family === "medium" || family === "large") return family;
  // extraLarge (iPad) and any future family get the large layout, which has the
  // most room and scales up without clipping.
  return family ? "large" : PREVIEW_SIZE;
}

/** "dark"/"clear" from the Parameter field, else the APPEARANCE constant. */
function resolveAppearance() {
  const tokens = parameterTokens();
  if (tokens.indexOf("clear") !== -1) return "clear";
  if (tokens.indexOf("dark") !== -1) return "dark";
  return APPEARANCE;
}

/**
 * `base` with `params` appended as a query string.
 *
 * Built by hand rather than with `URL`/`URLSearchParams`: those are Web APIs
 * supplied by browsers, and Scriptable runs on bare JavaScriptCore with no DOM,
 * so referencing them throws before a request is ever made.
 */
function buildUrl(base, params) {
  const parts = [];
  for (const key of Object.keys(params)) {
    const value = params[key];
    if (value === null || value === undefined || value === "") continue;
    parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(value)}`);
  }
  if (parts.length === 0) return base;
  return base + (base.indexOf("?") === -1 ? "?" : "&") + parts.join("&");
}

/**
 * Fetches today's numbers. Throws on any network, status or parse problem so
 * there's a single failure path for the caller to handle.
 */
async function loadData() {
  const url = buildUrl(API_URL, {
    person: resolvePerson(),
    tz: deviceTimeZone(),
    key: API_KEY,
  });

  const req = new Request(url);
  req.timeoutInterval = 10;
  req.headers = { accept: "application/json" };

  const data = await req.loadJSON();

  // loadJSON resolves for error responses too, so check the status ourselves.
  const status = req.response ? req.response.statusCode : 200;
  if (status < 200 || status >= 300) {
    throw new Error(`HTTP ${status}`);
  }
  if (!data || typeof data !== "object") {
    throw new Error("Malformed response");
  }
  return data;
}

/** Normalises the payload into what the view needs, tolerating missing fields. */
function toViewModel(data) {
  const rows = EXERCISES.map((ex) => {
    const value = safeNum(data[ex.key]);
    const goal = safeNum(data[ex.goalKey], FALLBACK_GOALS[ex.goalKey]);
    const percent = pct(value, goal);
    return {
      ...ex,
      value,
      goal,
      percent,
      done: value >= goal,
      remaining: Math.max(0, goal - value),
    };
  });

  const overall = rows.reduce((sum, r) => sum + r.percent, 0) / rows.length;
  const allComplete = rows.every((r) => r.done);

  return {
    rows,
    overall,
    allComplete,
    streak: safeNum(data.streak),
    name: typeof data.name === "string" ? data.name : "",
  };
}

/**
 * A rounded progress bar as an image.
 *
 * Drawn rather than stacked because a Scriptable stack can't clip a child to a
 * corner radius reliably at these sizes — the fill would poke out of the track.
 */
function progressBar(percent, hex, width = 300, height = 10) {
  const dc = new DrawContext();
  dc.size = new Size(width, height);
  dc.opaque = false;
  dc.respectScreenScale = true;

  const radius = height / 2;

  // Track.
  const track = new Path();
  track.addRoundedRect(new Rect(0, 0, width, height), radius, radius);
  dc.addPath(track);
  dc.setFillColor(THEME.track);
  dc.fillPath();

  // Fill. Kept at least as wide as it is tall so a tiny value still shows as a
  // dot instead of a sliver, and never wider than the track.
  const raw = (Math.max(0, Math.min(100, percent)) / 100) * width;
  if (raw > 0) {
    const fillWidth = Math.min(width, Math.max(height, raw));
    const fill = new Path();
    fill.addRoundedRect(new Rect(0, 0, fillWidth, height), radius, radius);
    dc.addPath(fill);
    dc.setFillColor(new Color(hex));
    dc.fillPath();
  }

  return dc.getImage();
}

/**
 * A circular progress ring with the percentage in the middle.
 *
 * DrawContext has no arc primitive, so the arc is a stroked path sampled at one
 * point per degree — smooth enough at this size.
 */
function ring(percent, hex, size = 54, lineWidth = 6) {
  const dc = new DrawContext();
  dc.size = new Size(size, size);
  dc.opaque = false;
  dc.respectScreenScale = true;

  const center = size / 2;
  const radius = center - lineWidth / 2;
  const clamped = Math.max(0, Math.min(100, percent));

  // Track ring.
  dc.setLineWidth(lineWidth);
  dc.setStrokeColor(THEME.track);
  dc.strokeEllipse(new Rect(lineWidth / 2, lineWidth / 2, size - lineWidth, size - lineWidth));

  // Progress arc, starting at 12 o'clock and sweeping clockwise.
  const sweep = Math.round((clamped / 100) * 360);
  if (sweep > 0) {
    const path = new Path();
    for (let deg = 0; deg <= sweep; deg++) {
      const rad = (deg - 90) * (Math.PI / 180);
      const point = new Point(center + radius * Math.cos(rad), center + radius * Math.sin(rad));
      if (deg === 0) path.move(point);
      else path.addLine(point);
    }
    dc.addPath(path);
    dc.setStrokeColor(new Color(hex));
    dc.setLineWidth(lineWidth);
    dc.strokePath();
  }

  // Centre label, sized from the ring so it stays proportional across families.
  const labelFont = Math.max(10, Math.round(size * 0.27));
  dc.setFont(Font.boldSystemFont(labelFont));
  dc.setTextColor(THEME.text);
  dc.setTextAlignedCenter();
  dc.drawTextInRect(
    `${Math.round(clamped)}%`,
    new Rect(0, center - labelFont * 0.72, size, labelFont * 1.5),
  );

  return dc.getImage();
}

/** The overall-progress ring as a widget image, with its clear-mode shadow. */
function addRing(stack, model, size, lineWidth) {
  const image = stack.addImage(
    ring(model.overall, model.allComplete ? "#30D158" : "#0A84FF", size, lineWidth),
  );
  image.imageSize = new Size(size, size);
  shadow(image);
  return image;
}

/** Shared chrome: background gradient, padding, tap target, refresh hint. */
function baseWidget(family) {
  const w = new ListWidget();

  const gradient = new LinearGradient();
  gradient.colors = [THEME.bg, THEME.bgGradientTo];
  gradient.locations = [0, 1];
  w.backgroundGradient = gradient;
  w.backgroundColor = THEME.bg;

  // Smaller widgets get tighter margins — 16pt of padding costs a Small widget
  // a fifth of its width.
  const pad = family === "small" ? 12 : family === "medium" ? 14 : 16;
  w.setPadding(pad, pad, pad - 2, pad);

  w.refreshAfterDate = new Date(Date.now() + REFRESH_MINUTES * 60 * 1000);
  if (TAP_URL) w.url = TAP_URL;

  return w;
}

/**
 * A soft drop shadow, applied only in clear mode. Returns the element so it can
 * wrap a call: `shadow(stack.addText("…"))`.
 *
 * Translucent styling puts white text and thin artwork over a wallpaper nobody
 * can predict. A shadow keeps both readable over a bright one without darkening
 * the scrim to the point that the glass effect is lost.
 */
function shadow(element) {
  if (!THEME.clear) return element;
  element.shadowColor = new Color("#000000", 0.55);
  element.shadowOffset = new Point(0, 1);
  element.shadowRadius = 3;
  return element;
}

/** Adds the "last updated" stamp, pinned to the bottom. */
function addFooter(w, family) {
  w.addSpacer();
  // A Small widget has no room for the label, and the time alone still answers
  // the only question the footer exists to answer: is this stale?
  const footer = w.addText(family === "small" ? nowHHmm() : `Last Updated: ${nowHHmm()}`);
  footer.font = Font.systemFont(family === "small" ? 9 : 10);
  footer.textColor = THEME.textDim;
  footer.centerAlignText();
  shadow(footer);
}

/** Large: title row, full-width bar, then percentage and what's left. */
function exerciseRow(w, row) {
  const container = w.addStack();
  container.layoutVertically();
  container.spacing = 5;

  // Title row: icon, label, ✅ when done, value / goal on the right.
  const head = container.addStack();
  head.layoutHorizontally();
  head.centerAlignContent();
  head.spacing = 5;

  const icon = head.addText(row.icon);
  icon.font = Font.systemFont(13);

  const label = head.addText(row.label);
  label.font = Font.semiboldSystemFont(13);
  label.textColor = THEME.text;
  shadow(label);

  if (row.done) {
    const check = head.addText("✅");
    check.font = Font.systemFont(11);
  }

  head.addSpacer();

  const value = head.addText(`${fmt(row.value)} / ${fmt(row.goal)}`);
  value.font = Font.mediumSystemFont(12);
  value.textColor = THEME.textDim;
  shadow(value);

  // Bar.
  const bar = container.addImage(progressBar(row.percent, row.color, 300, 10));
  bar.imageSize = new Size(300, 10);
  bar.applyFittingContentMode();
  shadow(bar);

  // Percentage + remaining.
  const meta = container.addStack();
  meta.layoutHorizontally();

  const percentText = meta.addText(`${Math.round(row.percent)}%`);
  percentText.font = Font.semiboldSystemFont(11);
  percentText.textColor = new Color(row.color);
  shadow(percentText);

  meta.addSpacer();

  const remaining = meta.addText(
    row.done ? "Goal reached" : `${fmt(row.remaining)} ${row.unit} to go`,
  );
  remaining.font = Font.systemFont(11);
  remaining.textColor = THEME.textDim;
  shadow(remaining);
}

  shadow(title);

  if (model.allComplete) {
    const trophy = titleRow.addText("🏆");
    trophy.font = Font.systemFont(12);
  }

  // The name shares the title's line, and the date is dropped entirely: Medium
  // is 158pt tall, and a second header line costs one of the three exercises.
  // A widget captioned "Today" doesn't need to spell the date out, and the
  // footer already timestamps it.
  titleRow.addSpacer();

  if (model.name) {
    const who = titleRow.addText(model.name);
    who.font = Font.systemFont(10);
    who.textColor = THEME.textDim;
    who.lineLimit = 1;
    shadow(who);
  }

  left.addSpacer(6);

  model.rows.forEach((row, i) => {
    if (i > 0) left.addSpacer(5);
    compactRow(left, row, 196);
  });

  body.addSpacer(10);

  // Right: overall ring above the streak.
  const right = body.addStack();
  right.layoutVertically();
  right.centerAlignContent();
  right.spacing = 6;

  addRing(right, model, 58, 6);
  streakBadge(right, model, false);

  const message = right.addText(motivation(model.overall, model.allComplete));
  message.font = Font.semiboldSystemFont(10);
  message.textColor = model.allComplete ? THEME.success : THEME.textDim;
  message.centerAlignText();
  message.lineLimit = 2;
  message.minimumScaleFactor = 0.7;
  shadow(message);

  addFooter(w, "medium");

  return w;
}

/**
 * Small: overall ring, streak, and three icon-and-bar rows.
 *
 * Everything that needs a sentence is dropped — the motivational line, the
 * date, the "to go" counts. What's left is what a Small widget is actually
 * glanced at for: am I on track, and is the streak alive.
 */
function buildSmall(model) {
  const w = baseWidget("small");

  const head = w.addStack();
  head.layoutHorizontally();
  head.centerAlignContent();

  const title = head.addText(model.allComplete ? "Today 🏆" : "Today");
  title.font = Font.boldSystemFont(13);
  title.textColor = THEME.text;
  title.lineLimit = 1;
  shadow(title);

  head.addSpacer();
  streakBadge(head, model, true);

  w.addSpacer(4);

  // The ring gets its own centred row so it isn't left-aligned under the title.
  const ringRow = w.addStack();
  ringRow.layoutHorizontally();
  ringRow.addSpacer();
  addRing(ringRow, model, 54, 6);
  ringRow.addSpacer();

  w.addSpacer(6);

  model.rows.forEach((row, i) => {
    if (i > 0) w.addSpacer(4);
    // 78pt leaves room for a three-digit "100%" on a 4.7" phone, where a Small
    // widget is only ~141pt wide.
    miniRow(w, row, 78);
  });

  addFooter(w, "small");

  return w;
}

/** The widget for a successful fetch, in whichever layout the family calls for. */
function buildWidget(data, family) {
  const model = toViewModel(data);
  if (family === "small") return buildSmall(model);
  if (family === "medium") return buildMedium(model);
  return buildLarge(model);
}

/**
 * Shown when the API can't be reached or answers with something unusable.
 *
 * `detail` is the underlying error, and is only ever passed when the script is
 * run by hand inside the app — a home-screen widget shows the plain message.
 * Diagnosing a widget you can't put a breakpoint in is otherwise guesswork.
 */
function buildErrorWidget(message, detail, family) {
  const small = family === "small";
  const w = baseWidget(family);

  w.addSpacer();

  const center = w.addStack();
  center.layoutVertically();
  center.centerAlignContent();

  const title = center.addText(small ? "Today" : "Today's Workout");
  title.font = Font.boldSystemFont(small ? 13 : 19);
  title.textColor = THEME.text;
  title.centerAlignText();
  title.lineLimit = 1;
  shadow(title);

  center.addSpacer(small ? 6 : 10);

  const icon = center.addText("⚠️");
  icon.font = Font.systemFont(small ? 22 : 28);
  icon.centerAlignText();

  center.addSpacer(small ? 6 : 8);

  // "Unable to load workout data." won't fit a Small widget at a readable size.
  const text = center.addText(small ? "No data" : message);
  text.font = Font.mediumSystemFont(small ? 11 : 14);
  text.textColor = THEME.textDim;
  text.centerAlignText();
  text.lineLimit = small ? 2 : 3;
  text.minimumScaleFactor = 0.85;
  shadow(text);

  // The detail is a raw error string; there's nowhere to put it on a Small.
  if (detail && !small) {
    center.addSpacer(8);
    const hint = center.addText(String(detail));
    hint.font = Font.systemFont(10);
    hint.textColor = THEME.streak;
    hint.centerAlignText();
    hint.lineLimit = 4;
    hint.minimumScaleFactor = 0.7;
    shadow(hint);
  }

  addFooter(w, family);
  return w;
}

// ---------------------------------------------------------------------------
// Entry point
// ---------------------------------------------------------------------------

async function main() {
  // Both depend on the Parameter field, so they're resolved before anything is
  // drawn — every builder below reads THEME.
  const family = resolveFamily();
  THEME = themeFor(resolveAppearance());

  let widget;

  try {
    const data = await loadData();
    widget = buildWidget(data, family);
  } catch (error) {
    console.error(`Widget failed: ${error}`);
    // The detail is suppressed on the home screen, where the brief message is
    // all there's room for, and shown when run by hand so it can be debugged.
    widget = buildErrorWidget(
      "Unable to load workout data.",
      config.runsInWidget ? null : String(error),
      family,
    );
  }

  if (config.runsInWidget) {
    Script.setWidget(widget);
  } else {
    // Running inside the Scriptable app — preview at the size being built.
    if (family === "small") await widget.presentSmall();
    else if (family === "medium") await widget.presentMedium();
    else await widget.presentLarge();
  }

  Script.complete();
}

await main();
